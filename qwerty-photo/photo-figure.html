{{- /* photo-figure is a post's photos, its byline (the author's avatar, name, and date), and its caption, shared by the full page (view.html) and the gallery's modal (modal.html). It is called as (array $builder $mode), where $mode is "page" or "modal". Several photos make a carousel: a track that slides one photo at a time, by its arrows, the Left and Right arrow keys, or a press-and-drag swipe. qwerty-theme's carousel bundle runs it (qwerty-theme/carousel/qwerty-carousel.js). Past the first or last photo, or on a post of one photo, the arrow keys go on to the previous or next post in the gallery's order: pinned posts first, then the rest, each newest first (qwerty-collection/items-gallery.html). Those posts are the hidden links at the foot of the carousel, which open as a modal or as a page, as this one did. Its CSS is in qwerty-outbox/stylesheet/collections.css, which the profile page loads, and which view.html links. */ -}} {{- $builder := index . 0 -}} {{- $mode := index . 1 -}} {{- $streamID := $builder.StreamID -}} {{- $photos := $builder.AttachmentsByCategory "photo" -}} {{- $canEdit := $builder.UserCan "edit" -}} {{- /* The caption is each photo's accessible name (aria-labelledby), so a post needs no separate alt text */ -}} {{- $captionID := print "photo-caption-" $mode "-" $streamID -}} {{- $hasCaption := ne "" (print $builder.ContentHTML) -}} {{- /* The title, when there is one, names each photo ahead of the caption */ -}} {{- $titleID := print "photo-title-" $mode "-" $streamID -}} {{- $hasTitle := ne "" $builder.Label -}} {{- $labelledBy := "" -}} {{- if $hasTitle -}}{{- $labelledBy = $titleID -}}{{- end -}} {{- if $hasCaption -}}{{- $labelledBy = trim (print $labelledBy " " $captionID) -}}{{- end -}} {{- $several := gt (len $photos) 1 -}} {{- /* The neighboring posts, in the gallery's order. Each lookup is one row: the pinned posts, then the rest. */ -}} {{- $gallery := $builder.Siblings.Where "templateId" "qwerty-photo" -}} {{- $pinned := $gallery.Where "isFeatured" true -}} {{- $unpinned := $gallery.Where "isFeatured" false -}} {{- $created := $builder.CreateDate -}} {{- $prev := "" -}} {{- $next := "" -}} {{- if $builder.IsFeatured -}} {{- $newer := (($pinned.WhereGT "createDate" $created).ByCreateDate.Top1.Slice).First -}} {{- $older := (($pinned.WhereLT "createDate" $created).ByCreateDate.Reverse.Top1.Slice).First -}} {{- if $older.ObjectID.IsZero -}} {{- $older = ($unpinned.ByCreateDate.Reverse.Top1.Slice).First -}} {{- end -}} {{- if not $newer.ObjectID.IsZero -}}{{- $prev = $newer.StreamID -}}{{- end -}} {{- if not $older.ObjectID.IsZero -}}{{- $next = $older.StreamID -}}{{- end -}} {{- else -}} {{- $newer := (($unpinned.WhereGT "createDate" $created).ByCreateDate.Top1.Slice).First -}} {{- if $newer.ObjectID.IsZero -}} {{- $newer = ($pinned.ByCreateDate.Top1.Slice).First -}} {{- end -}} {{- $older := (($unpinned.WhereLT "createDate" $created).ByCreateDate.Reverse.Top1.Slice).First -}} {{- if not $newer.ObjectID.IsZero -}}{{- $prev = $newer.StreamID -}}{{- end -}} {{- if not $older.ObjectID.IsZero -}}{{- $next = $older.StreamID -}}{{- end -}} {{- end -}}
