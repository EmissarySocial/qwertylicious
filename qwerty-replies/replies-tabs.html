{{- /* replies-tabs is the tab bar over a collection item: its own content first, then Replies. It is called as (array $builder $label $mode). $label names the first tab, and $mode is "page" (the item's page, whose tabs reload
and push ?tab=replies to the address bar) or "modal" (the photo modal, whose tabs re-open the modal through its modal action). The server picks the selected tab from ?tab=, as on the profile. There is no bar at all when the item neither accepts replies nor has any, so a private collection's items show none. An optional fourth value, "count", is a podcast episode's style. It labels the second tab with the number of replies on record (.ReplyCount): "0 Replies", "1 Reply", "3 Replies". And it leaves the bar out whenever replies are closed (.RepliesClosed, which an episode takes from its podcast), even if some arrived before they were closed. Without it the tab reads "Replies", with the number beside it in gray once there are any. */ -}} {{- $builder := index . 0 -}} {{- $label := index . 1 -}} {{- $mode := index . 2 -}} {{- $style := "" -}} {{- if gt (len .) 3 -}}{{- $style = index . 3 -}}{{- end -}} {{- $showBar := or $builder.AcceptsReplies (gt $builder.ReplyCount 0) -}} {{- if eq $style "count" -}} {{- if $builder.RepliesClosed -}}{{- $showBar = false -}}{{- end -}} {{- end -}} {{- if $showBar -}} {{- $repliesSelected := eq ($builder.QueryParam "tab") "replies" -}} {{- $base := print "/" $builder.StreamID -}} {{- if eq $mode "modal" -}} {{- $base = print $base "/modal" -}} {{- end -}} {{- $repliesURL := print $base "?tab=replies" -}}
{{$label}} {{- if eq $style "count" -}} {{- $builder.ReplyCount }} {{ if eq $builder.ReplyCount 1 }}Reply{{ else }}Replies{{ end -}} {{- else -}} Replies{{if gt $builder.ReplyCount 0}} {{$builder.ReplyCount}}{{end}} {{- end -}}
{{- end -}}
